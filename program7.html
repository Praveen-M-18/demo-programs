server.js
const express = require("express");
const mongoose = require("mongoose");
const app = express();

app.use(express.urlencoded({ extended: true }));

const employeeSchema = new mongoose.Schema({
    empid: String,
    name: String,
    department: String
});

const Employee = mongoose.model("Employee", employeeSchema);

app.get("/", (req, res) => {
    res.sendFile(__dirname + "/index.html");
});

app.post("/add", async (req, res) => {
    try {
        await Employee.create({
            empid: req.body.empid,
            name: req.body.name,
            department: req.body.department
        });
        res.redirect("/view");
    } catch (err) {
        res.status(500).send("Error adding employee: " + err.message);
    }
});

app.get("/view", async (req, res) => {
    try {
        const employees = await Employee.find();

        const escapeHTML = (value) => String(value).replace(/[&<>"']/g, char => ({
            "&": "&amp;",
            "<": "&lt;",
            ">": "&gt;",
            '"': "&quot;",
            "'": "&#39;"
        })[char]);

        let result = `<!DOCTYPE html>
        <html><head><title>Employees</title></head>
        <body><h2>Employee Details</h2>
        <table border="1" cellpadding="10">
        <tr><th>Employee ID</th>
        <th>Name</th><th>Department</th><th>Action</th></tr>`;

        employees.forEach(emp => {
            result += `<tr>
            <td>${escapeHTML(emp.empid)}</td>
            <td>${escapeHTML(emp.name)}</td>
            <td>${escapeHTML(emp.department)}</td>
            <td><a href="/delete/${emp._id}">Delete</a></td>
            </tr>`;
        });

        result += `</table><br><a href="/">Back</a></body></html>`;
        res.send(result);
    } catch (err) {
        res.status(500).send("Error retrieving employees: " + err.message);
    }
});

app.get("/delete/:id", async (req, res) => {
    try {
        if (!mongoose.isValidObjectId(req.params.id)) {
            return res.status(400).send("Invalid employee ID");
        }
        await Employee.findByIdAndDelete(req.params.id);
        res.redirect("/view");
    } catch (err) {
        res.status(500).send("Error deleting employee: " + err.message);
    }
});

mongoose.connect("mongodb://127.0.0.1:27017/EmployeeDB")
    .then(() => {
        console.log("Connected to MongoDB");
        app.listen(3000, () => {
            console.log("Server running at http://localhost:3000");
        });
    })
    .catch(err => {
        console.error("MongoDB connection failed:", err.message);
    });

📝 Important Notes — How to Run

1. Install the requirements

  • Install Node.js from https://nodejs.org/.
  • Install MongoDB Community Server and make sure the MongoDB service is running locally.
  • Check Node.js and npm by running node -v and npm -v.

2. Create the project

  • Create a folder named EmployeePortal.
  • Save index.html, server.js, and package.json inside it.
  • You do not need to create package-lock.json manually. npm creates the actual lock file when you install the dependencies.

3. Install the packages

Terminal commands
npm install
npm ls express mongoose

4. Start the application

Run command
npm start
  • Wait for the messages indicating that MongoDB is connected and the server is running.
  • Open http://localhost:3000 in your browser.
  • Enter an employee ID, name, and department, then click Add Employee.
  • Use View Employees to view records and Delete to remove a record.

5. Important reminders

  • MongoDB must be running before you start the application.
  • The database EmployeeDB and collection are created automatically by MongoDB/Mongoose when the first employee is saved.
  • The package-lock.json example above is only a placeholder. Running npm install generates the complete lock file. Do not use this placeholder with npm ci.
  • This lab uses a GET route for deletion to keep the demonstration simple. Production applications should use a DELETE request and appropriate protections.
