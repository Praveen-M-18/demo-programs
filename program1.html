01. Image Slideshow using JavaScript

Automatically changes images every 3 seconds using JavaScript.

index.html 
<!doctype html>
<html lang="en">
  <head>
    <meta charset="UTF-8" />
    <meta name="viewport" content="width=device-width, initial-scale=1.0" />
    <title>Image Slideshow Application</title>
    <link rel="stylesheet" href="./style.css" />
  </head>
  <body>
    <img src="./bat.jpg" alt="Slideshow image" id="slideshow_image" class="image" />

    <script src="./app.js"></script>
  </body>
</html>
app.js 
let images = ["./bat.jpg", "./bat1.jpg", "./bat3.jpg"];

let index = 0;

let slideshowImage = document.getElementById("slideshow_image");

function changeImage() {
  index++;

  if (index >= images.length) {
    index = 0;
  }

  slideshowImage.src = images[index];
}

setInterval(changeImage, 3000);
style.css 
body {
  margin: 0;
  display: flex;
  justify-content: center;
  align-items: center;
  min-height: 100vh;
}

.image {
  width: 80%;
  max-width: 700px;
  height: auto;
}

📝 Important Notes

  • Create three files: index.html, app.js, and style.css.
  • Download three images and Name it bat , bat1 , bat3
  • If an image does not appear, check its filename and folder location.
