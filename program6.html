06. Student Registration Form using Node.js and Express.js

Collect student details using an HTML form and process the submitted data with an Express.js server.

index.html 
<!doctype html>
<html lang="en">
  <head>
    <meta charset="UTF-8" />
    <meta name="viewport" content="width=device-width, initial-scale=1.0" />
    <title>Login Form</title>
  </head>
  <body>
    <h1>Form</h1>

    <form action="/login" method="post">
      <label for="name">Name:</label>
      <input type="text" name="name" id="name"
        placeholder="Enter your name" required />
      <br /><br />

      <label for="rollno">Roll No:</label>
      <input type="text" name="rollno" id="rollno"
        placeholder="Enter your roll number" required />
      <br /><br />

      <label for="department">Department:</label>
      <input type="text" name="department" id="department"
        placeholder="Enter your department" required />
      <br /><br />

      <button type="submit">Submit</button>
    </form>
  </body>
</html>
backend.js 
const express = require("express");
const app = express();

app.use(express.urlencoded({ extended: true }));

app.get("/", function (req, res) {
  res.sendFile(__dirname + "/index.html");
});

app.post("/login", function (req, res) {
  var name = req.body.name;
  var rollno = req.body.rollno;
  var department = req.body.department;

  // Escape user input before displaying it as HTML.
  function escapeHTML(value) {
    return String(value).replace(/[<>&"']/g, function (char) {
      return {
        "<": "&lt;",
        ">": "&gt;",
        "&": "&amp;",
        '"': "&quot;",
        "'": "&#39;"
      }[char];
    });
  }

  res.send(
    "<h1>Name: " + escapeHTML(name) + "</h1>" +
    "<p>Roll No: " + escapeHTML(rollno) + "</p>" +
    "<p>Department: " + escapeHTML(department) + "</p>"
  );
});

app.listen(3000, function () {
  console.log("Server Running on http://localhost:3000");
});

📝 Important Notes — How to Run

1. Install Node.js

  • Download Node.js from https://nodejs.org/ and install it if needed.
  • Open Command Prompt or the VS Code terminal.
  • Check the installation:
Terminal commands 
node -v
npm -v

2. Create the project folder

  • Create a folder named StudentRegistration.
  • Save index.html, backend.js, and package.json inside it.
  • You do not need to create package-lock.json manually. npm generates the complete lock file for your installed dependencies.

3. Install Express.js

  • Open a terminal inside the project folder.
  • Run the following commands. The install command creates or updates package-lock.json.
Install dependencies --copy and paste one by one-- 
npm install
npm ls express
if above commands does not work
alternative commands: 
npm.cmd install
    npm.cmd ls express
    npm.cmd start

4. Run the server

Start command 
npm start
  • Wait for the message Server Running on http://localhost:3000.
  • Open http://localhost:3000 in your browser.
  • Enter the student name, roll number, and department.
  • Click Submit. The server displays the submitted details.

5. Important reminders

  • Keep the terminal open while using the application.
  • To stop the server, press Ctrl + C in the terminal.
  • This program displays the submitted information; it does not save student records in a database.
  • The form action /login must match the Express POST route.
  • The package-lock.json shown above is only a placeholder. Running npm install generates the complete lock file, including Express and its dependencies. Do not use the placeholder with npm ci.
